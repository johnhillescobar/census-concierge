"""Revert each behaviour of 9485c97, and each stand-in revert that survived Gate 1, one at a time.

Run from the worktree root. Each file is restored from the bytes read before its mutation.
"""

import subprocess
import sys
from pathlib import Path

SCRIPT = Path("scripts/jira_gate_line.py")
TESTS = Path("api/tests/test_jira_gate_line.py")
UNUSABLE = "UNUSABLE = (ValueError, KeyError, TypeError, AttributeError)\n"

PRODUCT = [
    *[
        (f"UNUSABLE without {name}", [(UNUSABLE, UNUSABLE.replace(old, new))])
        for name, old, new in (
            ("ValueError", "(ValueError, ", "("),
            ("KeyError", " KeyError,", ""),
            ("TypeError", " TypeError,", ""),
            ("AttributeError", ", AttributeError", ""),
        )
    ],
    ("the dry run catches ValueError only", [(
        "        except UNUSABLE as exc:\n            unjudged",
        "        except ValueError as exc:\n            unjudged",
    )]),
    ("the reads before a write catch ValueError only", [(
        "except UNUSABLE as exc:  # a reply",
        "except ValueError as exc:  # a reply",
    )]),
    ("after the write AttributeError is not caught", [(
        "        *UNUSABLE,\n",
        "        ValueError,\n        KeyError,\n        TypeError,\n",
    )]),
    ("after the write only ValueError is caught of the unusable", [(
        "        *UNUSABLE,\n",
        "        ValueError,\n",
    )]),
    ("a KeyError is shown as the bare name", [(
        '    if isinstance(exc, ValueError):\n        return str(exc)\n',
        "    if True:\n        return str(exc)\n",
    )]),
    ("a status category that is not text reads as open", [(
        "    if not isinstance(category, str):  #",
        "    if False:  #",
    )]),
    ("history entries need not be a list", [(
        "        if not isinstance(entries, list):\n",
        "        if False:\n",
    )]),
    ("history items need not be a list", [(
        "            if not isinstance(items, list):\n",
        "            if False:\n",
    )]),
    ("a history that ends early is taken as complete", [(
        '        if start >= page["total"]:\n',
        '        if not entries or start >= page["total"]:\n',
    )]),
]

STAND_IN = [
    ("an empty description reads as empty text, not null", [(
        "    return None if body is None else wiki(body)\n",
        '    return "" if body is None else wiki(body)\n',
    )]),
    ("the search returns descriptions", [(
        'return [{"id": "1", "key": key, "fields": {}} for key in self.tickets]',
        'return [{"id": "1", "key": key, "fields": {"description": body}} '
        "for key, body in self.tickets.items()]",
    )]),
    ("the search accepts whatever fields are asked for", [(
        '        assert fields == "status"\n', "",
    )]),
    ("history ids are made from the key and the count", [(
        '                "id": str(self.entries),\n',
        '                "id": f"{key}-{len(self.history.get(key, []))}",\n',
    )]),
    ("the history hands out its own entries", [(
        "values = copy.deepcopy(entries[start : start + self.page])",
        "values = entries[start : start + self.page]",
    )]),
    ("an issue reply has no id", [(
        'return {"id": "1", "key": self.moved.get(key, key), "fields": fields}',
        'return {"key": self.moved.get(key, key), "fields": fields}',
    )]),
    ("a search result has no id and no fields", [(
        'return [{"id": "1", "key": key, "fields": {}} for key in self.tickets]',
        'return [{"key": key} for key in self.tickets]',
    )]),
    ("a history page has only values and total", [(
        '                "startAt": start,\n                "maxResults": self.page,\n',
        "",
    ), (
        '                "isLast": start + len(values) >= len(entries),\n', "",
    )]),
    ("a history entry has no author and no created", [(
        '                "author": {"accountId": "someone"},\n'
        '                "created": "2026-01-01T00:00:00.000+0000",\n',
        "",
    )]),
    ("a history item has only fieldId and the two texts", [(
        '            "field": field,\n            "fieldtype": "jira",\n', "",
    ), (
        '            "from": None,\n', "",
    ), (
        '            "to": None,\n', "",
    )]),
]


def run() -> tuple[int, str]:
    done = subprocess.run(
        [sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", str(TESTS)],
        capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=180,
    )
    lines = done.stdout.strip().splitlines()
    failed = sorted({
        line.split("::", 1)[1].split("[")[0].split(" ")[0]
        for line in lines if line.startswith("FAILED ")
    })
    return done.returncode, f"{lines[-1]}\n          " + "\n          ".join(failed[:6])


def main() -> int:
    code, summary = run()
    print(f"BASELINE (nothing reverted) {summary.strip()}")
    if code != 0:
        return 2
    bad = 0
    for title, path, mutations in (
        ("product, 9485c97", SCRIPT, PRODUCT),
        ("stand-in survivors of gate1-r5", TESTS, STAND_IN),
    ):
        print(f"\n== {title}")
        good = 0
        for label, swaps in mutations:
            original = path.read_bytes()
            text = original.decode("utf-8").replace("\r\n", "\n")
            for old, new in swaps:
                # the first occurrence is the stand-in's; later ones are test data
                assert old in text, (label, old)
                if path == SCRIPT:
                    assert text.count(old) == 1, (label, old)
                text = text.replace(old, new, 1)
            try:
                path.write_bytes(text.encode("utf-8"))
                code, summary = run()
            finally:
                path.write_bytes(original)
            # exit 1 is failing tests; 2 and up would be a broken mutation, not a caught one
            verdict = "GOOD" if code == 1 else f"BAD (exit {code})"
            good += code == 1
            bad += code != 1
            print(f"REVERTED: {label}\n  {verdict}\n        {summary}")
        print(f"SUBTOTAL: {len(mutations)} reverts, {good} GOOD, {len(mutations) - good} BAD")
    code, summary = run()
    print(f"\nRESTORED {summary.strip()}")
    return 1 if bad or code else 0


if __name__ == "__main__":
    sys.exit(main())
