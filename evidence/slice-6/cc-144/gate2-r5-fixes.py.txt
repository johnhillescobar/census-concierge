"""Revert each behaviour of the Gate 2 round fix, one at a time.

Run from the worktree root. The script is restored from the bytes read before its mutation.
"""

import subprocess
import sys
from pathlib import Path

SCRIPT = Path("scripts/jira_gate_line.py")
TESTS = Path("api/tests/test_jira_gate_line.py")

MUTATIONS = [
    # one ticket that cannot be judged
    ("only the top of a description is checked", [(
        '    content = value.get("content", [])\n'
        "    return isinstance(content, list) and all(_is_node(child) for child in content)\n",
        "    return True\n",
    )]),
    ("a content that is not a list passes", [(
        "    return isinstance(content, list) and all(_is_node(child) for child in content)\n",
        "    return not isinstance(content, list) or all(_is_node(child) for child in content)\n",
    )]),
    ("text that cannot be stored is not tried while judging", [(
        "            _dump(description)  # what cannot be stored", "            pass  #",
    )]),
    ("tickets are judged together, outside the per-ticket handler", [(
        '            found += changes([{"key": key, "fields": {"description": description}}])\n',
        "            held.append({\"key\": key, \"fields\": {\"description\": description}})\n",
    ), (
        "        judged += 1\n    return judged, found, unjudged\n",
        "        judged += 1\n    return judged, changes(held), unjudged\n",
    ), (
        "    found: list[Change] = []\n    unjudged",
        "    held: list[dict[str, Any]] = []\n    found: list[Change] = []\n    unjudged",

    ("a dry run refuses a ticket with no answer and goes on", [(

        '            raise ConnectionError(f"{key}: {_why(exc)}") from exc\n',
        "        except NO_ANSWER as exc:\n"

        "    if isinstance(exc, NO_ANSWER):\n", "    if False:\n",

def run() -> tuple[int, str]:
    done = subprocess.run(
        [sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider", str(TESTS)],
        capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=180,
    )
    lines = done.stdout.strip().splitlines()
    failed = sorted({
        line.split("::", 1)[1].split("[")[0].split(" ")[0]
        for line in lines if line.startswith("FAILED ")
    })
    return done.returncode, f"{lines[-1]}\n          " + "\n          ".join(failed[:5])


def main() -> int:
    code, summary = run()
    print(f"BASELINE (nothing reverted) {summary.strip()}")
    if code != 0:
        return 2
    good = bad = 0
    for label, swaps in MUTATIONS:
        original = SCRIPT.read_bytes()
        text = original.decode("utf-8").replace("\r\n", "\n")
        for old, new in swaps:
            assert text.count(old) == 1, (label, old)
            text = text.replace(old, new)
        compile(text, str(SCRIPT), "exec")  # a mutation that does not compile proves nothing
        try:
            SCRIPT.write_bytes(text.encode("utf-8"))
            code, summary = run()
        finally:
            SCRIPT.write_bytes(original)
        # exit 1 is failing tests; any other non-zero exit is a broken mutation, not a caught one
        verdict = "GOOD" if code == 1 else f"BAD (exit {code})"
        good += code == 1
        bad += code != 1
        print(f"REVERTED: {label}\n  {verdict}\n        {summary}")
    code, summary = run()
    print(f"TOTAL: {len(MUTATIONS)} reverts, {good} GOOD, {bad} BAD")
    print(f"RESTORED {summary.strip()}")
    return 1 if bad or code else 0


if __name__ == "__main__":
    sys.exit(main())
