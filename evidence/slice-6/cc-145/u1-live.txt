"""U1, live and read-only: do the last N comments jira_fetch prints equal the newest N, oldest first?"""

import sys

sys.path.insert(0, "scripts")
import jira_fetch as jf  # noqa: E402

jf.load_env()
for key, n in (("CC-144", 3), ("CC-145", 2)):
    every = jf._api("GET", f"/issue/{key}/comment?maxResults=100").get("comments", [])
    created = sorted(c["created"] for c in every)
    last = [c["created"] for c in jf._comments(key, n)]
    print(f"{key}: {len(every)} comments; oldest {created[0]}; newest {created[-1]}")
    print(f"  _comments({n}) created: {last}")
    print(f"  newest {n}, oldest first: {created[-n:]}")
    print("  MATCH" if last == created[-n:] else "  MISMATCH")
