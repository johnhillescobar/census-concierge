def run(command: str) -> tuple[int, str]:
    # split here, no shell: cmd.exe would pass the single quotes through to pytest's -k
    done = subprocess.run(
        shlex.split(command), cwd=ROOT, capture_output=True, text=True, encoding="utf-8", errors="replace"
    )
    lines = [ln for ln in (done.stdout + done.stderr).splitlines() if "VIRTUAL_ENV" not in ln]
    return done.returncode, "\n".join(lines)


def main() -> int:
    head = subprocess.run(["git", "rev-parse", "HEAD"], cwd=ROOT, capture_output=True, text=True).stdout.strip()
    wanted = sys.argv[1:] or list(ROWS)
    worst = 0
    for key in wanted:
        parts = [f"CC-145 {key}, HEAD {head}, {datetime.now(UTC).isoformat(timespec='seconds')}"]
        for command in ROWS[key]:
            code, output = run(command)
            worst = max(worst, code if not command.startswith("git grep") else 0)
            parts += ["", f"$ {command}", output, f"exit {code}"]
        (OUT / f"{key}.txt").write_bytes(("\n".join(parts) + "\n").encode("utf-8"))
        print(key, "written")
    return worst


if __name__ == "__main__":
    sys.exit(main())
