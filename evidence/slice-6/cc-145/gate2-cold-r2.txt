def text(value: str) -> dict[str, Any]:
    return {"type": "text", "text": value}


def para(*inline: Any) -> dict[str, Any]:
    return {"type": "paragraph", "content": [text(i) if isinstance(i, str) else i for i in inline]}


BREAK = {"type": "hardBreak"}


def doc(*blocks: dict[str, Any]) -> dict[str, Any]:
    return {"type": "doc", "version": 1, "content": list(blocks)}


def item(*blocks: dict[str, Any]) -> dict[str, Any]:
    return {"type": "listItem", "content": list(blocks)}


def bullets(*items: dict[str, Any]) -> dict[str, Any]:
    return {"type": "bulletList", "content": list(items)}


def heading(value: str, level: int = 3) -> dict[str, Any]:
    return {"type": "heading", "attrs": {"level": level}, "content": [text(value)]}


def cell(*blocks: dict[str, Any]) -> dict[str, Any]:
    return {"type": "tableCell", "content": list(blocks)}


def row(*cells: dict[str, Any]) -> dict[str, Any]:
    return {"type": "tableRow", "content": list(cells)}


def approved(description: object) -> dict[str, Any]:
    return {
        "approved_ids": closure.required_ids(closure.adf_text(description)),
        "approved_sha256": closure.criteria_fingerprint(description),
