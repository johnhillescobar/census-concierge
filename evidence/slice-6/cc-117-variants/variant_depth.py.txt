import sys, tomllib, collections
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0, "scripts"); sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv()
import eval_retrieval as ev
from src.retrieval.index import load
from src.retrieval import rerank
from src.retrieval.rerank import Candidate, choose
vp, depth, reps = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
head = rerank.PROMPT.split("Sibling tables share")[0]
body = open(vp, encoding="utf-8").read().strip()
if body != "CURRENT": rerank.PROMPT = head + body + '\n\nReturn JSON: {"table": "Bxxxxx"}'
qs = [q for q in tomllib.load(open("evals/golden_questions.toml", "rb"))["question"]
      if q.get("expect_table") and not q.get("holdout") and q.get("tier") in ("long_tail", "trap")]
search = ev.load_retriever(False); idx = load(); pos = {t: i for i, t in enumerate(idx.tables)}
pools = {q["id"]: list(search(q["text"], depth)) for q in qs}
def one(a):
    q, _ = a; r = pools[q["id"]]
    c = [Candidate(t, idx.titles[pos[t]], idx.universes[pos[t]]) for t in r]
    return q["id"], q["tier"], choose(q["text"], c) == q["expect_table"]
with ThreadPoolExecutor(8) as ex: res = list(ex.map(one, [(q, r) for r in range(reps) for q in qs]))
out = {}
for t in ("long_tail", "trap"):
    rr = [ok for i, tt, ok in res if tt == t]; out[t] = sum(rr) / len(rr)
miss = collections.Counter(i for i, t, ok in res if not ok)
print(f"{vp.split('/')[-1]} depth {depth}: long_tail {out['long_tail']:.3f}  trap {out['trap']:.3f} | misses/{reps}: {dict(sorted(miss.items()))}")
