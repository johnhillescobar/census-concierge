import sys, json, collections
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv(".env")
from src.retrieval import rerank
from src.retrieval.rerank import Candidate, choose
pools = json.load(open(sys.argv[1])); reps = int(sys.argv[2]); variants = sys.argv[3:]
head = rerank.PROMPT.split("Sibling tables share")[0]
def run(path):
    body = open(path, encoding="utf-8").read().replace("\\n", "").strip()
    prompt = head + body + '\n\nReturn JSON: {"table": "Bxxxxx"}'
    def one(a):
        qid, _ = a; p = pools[qid]
        rerank.PROMPT = prompt
        return qid, choose(p["text"], [Candidate(*c) for c in p["cands"]]) == p["expect"]
    jobs = [(i, r) for r in range(reps) for i in pools]
    with ThreadPoolExecutor(8) as ex: res = list(ex.map(one, jobs))
    return res
for v in variants:
    res = run(v)
    def rate(f):
        sel = [ok for i, ok in res if f(pools[i])]; return sum(sel) / len(sel), len(sel)
    lt = rate(lambda p: p["tier"] == "long_tail" and not p["holdout"]); tr = rate(lambda p: p["tier"] == "trap"); ho = rate(lambda p: p["holdout"]); co = rate(lambda p: p["tier"] == "core")
    miss = collections.Counter(i for i, ok in res if not ok)
    print(f"{v.split('/')[-1]:8} long_tail {lt[0]:.3f}  trap {tr[0]:.3f}  core {co[0]:.3f}  holdout {ho[0]:.3f} | misses/{reps}: {dict(sorted(miss.items()))}", flush=True)
