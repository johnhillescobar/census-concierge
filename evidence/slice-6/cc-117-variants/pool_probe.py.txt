import sys, tomllib, collections
sys.path.insert(0, "scripts"); sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv()
import eval_retrieval as ev
from src.retrieval.index import load
from src.retrieval.rerank import Candidate, choose
ids = sys.argv[2:]; reps = int(sys.argv[1])
qs = {q["id"]: q for q in tomllib.load(open("evals/golden_questions.toml", "rb"))["question"]}
search = ev.load_retriever(False); idx = load(); pos = {t: i for i, t in enumerate(idx.tables)}
for i in ids:
    q = qs[i]; ranked = list(search(q["text"], 10))
    cands = [Candidate(t, idx.titles[pos[t]], idx.universes[pos[t]]) for t in ranked]
    picks = collections.Counter(choose(q["text"], cands) for _ in range(reps))
    exp = q["expect_table"]
    print(f"{i} exp={exp} in_pool={exp in ranked} rank={ranked.index(exp)+1 if exp in ranked else None} picks={dict(picks)}")
    print("   pool:", [f"{t}:{idx.titles[pos[t]][:34]}" for t in ranked[:6]])
