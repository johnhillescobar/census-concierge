import sys, tomllib, collections, json
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0, "scripts"); sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv()
import eval_retrieval as ev
from src.retrieval.index import load
from src.retrieval import rerank
from src.retrieval.rerank import Candidate, choose
variant_path, reps, tier = sys.argv[1], int(sys.argv[2]), sys.argv[3]
head = rerank.PROMPT.split("Sibling tables share")[0]
body = open(variant_path, encoding="utf-8").read().strip()
if body != "CURRENT":
    rerank.PROMPT = head + body + '\n\nReturn JSON: {"table": "Bxxxxx"}'
qs = [q for q in tomllib.load(open("evals/golden_questions.toml", "rb"))["question"]
      if q.get("expect_table") and q.get("tier") == tier and bool(q.get("holdout")) == (tier == "holdout")]
search = ev.load_retriever(False); idx = load(); pos = {t: i for i, t in enumerate(idx.tables)}
pools = {q["id"]: list(search(q["text"], 10)) for q in qs}
def one(a):
    q, _ = a; ranked = pools[q["id"]]
    cands = [Candidate(t, idx.titles[pos[t]], idx.universes[pos[t]]) for t in ranked]
    return q["id"], choose(q["text"], cands) == q["expect_table"]
jobs = [(q, r) for r in range(reps) for q in qs]
with ThreadPoolExecutor(8) as ex: res = list(ex.map(one, jobs))
miss = collections.Counter(i for i, ok in res if not ok)
rate = 1 - len([1 for i, ok in res if not ok]) / len(res)
print(f"{variant_path.split('/')[-1]}: selector@1 = {rate:.3f} over {len(res)} picks; misses/{reps}: {dict(sorted(miss.items()))}")
