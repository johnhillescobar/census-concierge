import sys, tomllib, collections
sys.path.insert(0, "scripts"); sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv()
import eval_retrieval as ev
from src.retrieval.index import load
from src.retrieval.rerank import Candidate, choose
qs = {q["id"]: q for q in tomllib.load(open("evals/golden_questions.toml", "rb"))["question"]}
search = ev.load_retriever(False); idx = load(); pos = {t: i for i, t in enumerate(idx.tables)}
for i in sys.argv[1:]:
    q = qs[i]; exp = q["expect_table"]; ranked = list(search(q["text"], 10))
    cands = [Candidate(t, idx.titles[pos[t]], idx.universes[pos[t]]) for t in ranked]
    picks = collections.Counter(choose(q["text"], cands) for _ in range(3))
    print(f"\n=== {i}  expected {exp}  ({'rank %d' % (ranked.index(exp)+1) if exp in ranked else 'NOT IN TOP 10'})")
    print(f"Q: {q['text']}")
    print(f"selector picks x3: {dict(picks)}")
    for n, t in enumerate(ranked, 1):
        mark = "  <-- EXPECTED" if t == exp else ("  <-- PICKED" if t in picks else "")
        print(f"  {n:2}. {t:8} {idx.titles[pos[t]][:78]} | {idx.universes[pos[t]][:42]}{mark}")
    if exp not in ranked:
        print(f"  expected table is: {exp}  {idx.titles[pos[exp]]} | {idx.universes[pos[exp]]}")
