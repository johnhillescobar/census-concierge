import sys, json, tomllib, collections
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0, "scripts"); sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv()
import eval_retrieval as ev
from src.retrieval.index import load
from src.retrieval.rerank import Candidate, choose
from src.tools import SearchTablesInput
from openai import OpenAI
S = 3
desc = SearchTablesInput.model_fields["topic"].description
qs = [q for q in tomllib.load(open("evals/golden_questions.toml", "rb"))["question"]
      if q.get("expect_table") and not q.get("holdout") and q.get("tier") in ("long_tail", "trap", "core")]
client = OpenAI()
def topics(q):
    out = []
    for _ in range(S):
        r = client.chat.completions.create(model="gpt-4o-mini", temperature=0.8,
            messages=[{"role": "system", "content": "You call a search tool. Write only the value of its `topic` argument. Argument description: " + desc},
                      {"role": "user", "content": q["text"]}])
        out.append(r.choices[0].message.content.strip().strip("'\""))
    return q["id"], out
with ThreadPoolExecutor(4) as ex: T = dict(ex.map(topics, qs))
json.dump(T, open(sys.argv[1], "w"), indent=1)
search = ev.load_retriever(False); idx = load(); pos = {t: i for i, t in enumerate(idx.tables)}
def rrf(a, b, k=60):
    sc = {}
    for lst in (a, b):
        for r, t in enumerate(lst, 1): sc[t] = sc.get(t, 0) + 1 / (k + r)
    return [t for t, _ in sorted(sc.items(), key=lambda x: -x[1])]
def pool(strategy, q, topic):
    full = list(search(q["text"], 30)); top = list(search(topic, 30))
    if strategy == "Q": return full
    if strategy == "T": return top
    if strategy == "F": return rrf(top, full)[:30]
    if strategy == "U":
        p = top[:20]
        return p + [t for t in full if t not in p][: 30 - len(p)]
jobs = [(st, q, s) for st in ("Q", "T", "U", "F") for q in qs for s in range(S)]
def one(a):
    st, q, s = a; p = pool(st, q, T[q["id"]][s])
    ok = choose(q["text"], [Candidate(t, idx.titles[pos[t]], idx.universes[pos[t]]) for t in p]) == q["expect_table"]
    inpool = q["expect_table"] in p
    return st, q["id"], q["tier"], ok, inpool
with ThreadPoolExecutor(4) as ex: res = list(ex.map(one, jobs))
for st in ("Q", "T", "U", "F"):
    row = {t: [r for r in res if r[0] == st and r[2] == t] for t in ("long_tail", "trap", "core")}
    msg = "  ".join(f"{t} {sum(r[3] for r in v)/len(v):.3f} (in-pool {sum(r[4] for r in v)/len(v):.3f})" for t, v in row.items())
    miss = collections.Counter(r[1] for r in res if r[0] == st and not r[3])
    print(f"{st}: {msg} | misses/3: {dict(sorted(miss.items()))}")
