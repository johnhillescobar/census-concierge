import sys, tomllib, collections
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0, "scripts"); sys.path.insert(0, "api")
from dotenv import load_dotenv; load_dotenv()
import eval_retrieval as ev
from src.retrieval.index import load
from src.retrieval.rerank import Candidate, choose
depths = [int(x) for x in sys.argv[1].split(",")]; reps = int(sys.argv[2])
allq = [q for q in tomllib.load(open("evals/golden_questions.toml", "rb"))["question"] if q.get("expect_table") and not q.get("holdout")]
search = ev.load_retriever(False); idx = load(); pos = {t: i for i, t in enumerate(idx.tables)}
pools = {q["id"]: list(search(q["text"], max(depths))) for q in allq}
def one(a):
    q, d, _ = a; ranked = pools[q["id"]][:d]
    if not ranked: return q["id"], d, False
    cands = [Candidate(t, idx.titles[pos[t]], idx.universes[pos[t]]) for t in ranked]
    return q["id"], d, choose(q["text"], cands) == q["expect_table"]
tier = {q["id"]: q.get("tier") for q in allq}
jobs = [(q, d, r) for d in depths for r in range(reps) for q in allq]
with ThreadPoolExecutor(8) as ex: res = list(ex.map(one, jobs))
for d in depths:
    for t in ("long_tail", "trap", "core"):
        rr = [ok for i, dd, ok in res if dd == d and tier[i] == t]
        print(f"depth {d:2} {t:9} selector@1 = {sum(rr)/len(rr):.3f} ({len(rr)} picks)")
    miss = collections.Counter(i for i, dd, ok in res if dd == d and not ok)
    print(f"depth {d:2} misses/{reps}: {dict(sorted(miss.items()))}")
